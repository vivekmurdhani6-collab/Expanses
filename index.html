/** * LEDGER — Apps Script backend (JSON body, one tab per month) * Paste this whole file into: your Google Sheet → Extensions → Apps Script * (delete whatever boilerplate code is already there first) * * Set your own SECRET_KEY below before deploying — it must match the PIN * you type into the web page. */ const SECRET_KEY = "change-this-to-your-own-passcode"; const TEMPLATE_SHEET_NAME = "Template"; // must match exactly — never rename/edit this tab function doPost(e) { try { const data = JSON.parse(e.postData.contents); if (data.pin !== SECRET_KEY) { return jsonResponse({ result: "error", error: "Incorrect PIN" }); } if (!data.date || !data.type || !data.category || !data.amount) { return jsonResponse({ result: "error", error: "Missing a required field" }); } const entryDate = new Date(data.date); const monthLabel = Utilities.formatDate(entryDate, Session.getScriptTimeZone(), "MMM-yyyy"); const ss = SpreadsheetApp.getActiveSpreadsheet(); let sheet = ss.getSheetByName(monthLabel); if (!sheet) { const template = ss.getSheetByName(TEMPLATE_SHEET_NAME); if (!template) { return jsonResponse({ result: "error", error: "Template tab not found — don't rename it" }); } sheet = template.copyTo(ss); sheet.setName(monthLabel); ss.setActiveSheet(sheet); ss.moveActiveSheet(template.getIndex() + 1); } const lastRow = sheet.getLastRow(); let targetRow = null; const rowsToCheck = Math.max(lastRow - 1, 1); const colA = sheet.getRange(2, 1, rowsToCheck, 1).getValues(); for (let i = 0; i < colA.length; i++) { if (!colA[i][0]) { targetRow = 2 + i; break; } } if (!targetRow) targetRow = lastRow + 1; // ran out of pre-formatted rows — append a fresh one sheet.getRange(targetRow, 1, 1, 5).setValues([[ entryDate, data.type, data.category, data.description || "", Number(data.amount) ]]); return jsonResponse({ result: "success", sheet: monthLabel, row: targetRow }); } catch (err) { return jsonResponse({ result: "error", error: err.toString() }); } } function doGet(e) { return jsonResponse({ status: "Ledger API is running" }); } function jsonResponse(obj) { return ContentService .createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
